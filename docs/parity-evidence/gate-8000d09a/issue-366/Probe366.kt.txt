private const val SINGLE = "visual_export_qualification"

fun probe366ViewController(multiOnly: Boolean): UIViewController = ComposeUIViewController(configure = {
    onFocusBehavior = androidx.compose.ui.uikit.OnFocusBehavior.DoNothing
}) {
    MaterialTheme {
        val focus = LocalFocusManager.current
        if (multiOnly) {
            androidx.compose.foundation.layout.Row(Modifier.fillMaxSize().padding(horizontal = 8.dp)) {
                Button(onClick = { focus.clearFocus() }, modifier = Modifier.testTag("compose-reset")) { Text("R") }
                var m by remember { mutableStateOf(MULTI) }
                OutlinedTextField(m, { m = it }, textStyle = TextStyle(fontFamily = FontFamily.Monospace),
                    modifier = Modifier.weight(1f).height(150.dp).testTag("compose-multiline"))
            }
            return@MaterialTheme
        }
        Column(Modifier.fillMaxSize().padding(horizontal = 8.dp)) {
            Button(onClick = { focus.clearFocus() }, modifier = Modifier.testTag("compose-reset")) { Text("compose-reset") }
            var a by remember { mutableStateOf(SINGLE) }
            OutlinedTextField(a, { a = it }, singleLine = true, modifier = Modifier.fillMaxWidth().testTag("compose-outlined"))
            var b by remember { mutableStateOf(SINGLE) }
            BasicTextField(b, { b = it }, singleLine = true, textStyle = TextStyle(fontSize = 16.sp),
                modifier = Modifier.fillMaxWidth().padding(top = 8.dp).height(44.dp).border(1.dp, Color.Gray).testTag("compose-basic"))
            val s = rememberTextFieldState(SINGLE)
            BasicTextField(state = s, textStyle = TextStyle(fontSize = 16.sp),
                modifier = Modifier.fillMaxWidth().padding(top = 8.dp).height(44.dp).border(1.dp, Color.Gray).testTag("compose-state"))
            var m by remember { mutableStateOf(MULTI) }
            OutlinedTextField(m, { m = it }, textStyle = TextStyle(fontFamily = FontFamily.Monospace),
                modifier = Modifier.fillMaxWidth().padding(top = 8.dp).height(130.dp).testTag("compose-multiline"))
        }

}
